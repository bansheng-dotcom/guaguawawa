### EDA 的工程化实现——标准化的 run_eda() 函数

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from pathlib import Path

def run_eda(df: pd.DataFrame, output_dir: str = "output") -> dict:
    """标准化 EDA 流程——输出结构化报告。

    输出文件:
    - eda_summary.csv: 每列的数据类型/缺失率/唯一值数/取值示例
    - eda_distributions.png: 关键数值列的分布直方图
    - eda_correlations.png: 数值列的相关性热力图
    - eda_report.md: EDA 发现摘要（结论先行）

    Returns:
        dict with keys: n_rows, n_cols, missing_cols, skewed_cols, key_findings
    """
    out = Path(output_dir)
    out.mkdir(exist_ok=True)

    # 1. 列级摘要
    summary = pd.DataFrame({
        'column': df.columns,
        'dtype': df.dtypes.values,
        'missing_pct': (df.isna().mean() * 100).round(1).values,
        'unique_values': df.nunique().values,
        'sample_value': [str(df[col].dropna().iloc[0]) if not df[col].dropna().empty else 'N/A' for col in df.columns]
    })
    summary.to_csv(out / 'eda_summary.csv', index=False)

    # 2. 数值列分布
    num_cols = df.select_dtypes(include='number').columns[:9]  # 最多 9 列——3x3 子图
    if len(num_cols) > 0:
        fig, axes = plt.subplots(3, 3, figsize=(14, 12))
        for ax, col in zip(axes.flat, num_cols):
            df[col].dropna().hist(bins=30, ax=ax, edgecolor='white')
            ax.set_title(col, fontsize=10)
        fig.tight_layout()
        fig.savefig(out / 'eda_distributions.png', dpi=150)

    # 3. 相关性热力图
    if len(num_cols) >= 3:
        fig, ax = plt.subplots(figsize=(10, 8))
        corr = df[num_cols].corr()
        sns.heatmap(corr, annot=True, fmt='.2f', cmap='RdBu_r', center=0, ax=ax)
        ax.set_title('Feature Correlations')
        fig.tight_layout()
        fig.savefig(out / 'eda_correlations.png', dpi=150)

    # 4. 关键发现——高缺失率列 + 高偏度列
    high_missing = summary[summary['missing_pct'] > 20]['column'].tolist()
    skewed_cols = []
    for col in num_cols:
        skew = df[col].dropna().skew()
        if abs(skew) > 2:
            skewed_cols.append(f"{col} (skew={skew:.1f})")

    findings = {
        'n_rows': len(df),
        'n_cols': len(df.columns),
        'high_missing_cols': high_missing,
        'skewed_cols': skewed_cols
    }

    # 写 EDA 报告——结论先行
    with open(out / 'eda_report.md', 'w') as f:
        f.write(f"# EDA 发现摘要\n\n")
        f.write(f"**数据规模:** {findings['n_rows']} 行 × {findings['n_cols']} 列\n\n")
        if high_missing:
            f.write(f"**高缺失率列:** {', '.join(high_missing)}——缺失率 > 20%——后续分析需注意\n\n")
        if skewed_cols:
            f.write(f"**高偏度列:** {', '.join(skewed_cols)}——这些列可能需要 log 变换\n\n")
        f.write("![分布图](eda_distributions.png)\n\n")
        f.write("![相关性热力图](eda_correlations.png)\n")

    return findings
```

**这个函数的价值——项目 1 写好了——项目 2、3 直接 import 复用——改 output_dir 参数即可。** 工程化的本质——不是"每个项目重新写一遍"——是"一次写好——到处复用"。

### 渠道 LTV 分析的完整工程实现

```python
def compute_channel_ltv(orders, channels, ltv_window_days=180):
    """计算各渠道 LTV（用户生命周期价值）。
    口径: 分子=SUM(amount) WHERE order_date BETWEEN first_order AND first_order+ltv_window_days
          分母=COUNT(DISTINCT user_id)——该渠道获取的总用户数
          时间窗口=180 天——行业标准——6 个月覆盖大部分用户活跃周期
    """
    first_order = orders.groupby("user_id").agg(
        first_date=("order_date","min"), channel=("channel","first")
    ).reset_index()
    merged = orders.merge(first_order, on="user_id")
    ltv_orders = merged[
        (merged["order_date"] >= merged["first_date"]) &
        (merged["order_date"] <= merged["first_date"] + pd.Timedelta(days=ltv_window_days))
    ]
    ltv_by_channel = ltv_orders.groupby("channel").agg(
        total_revenue=("amount","sum"), n_users=("user_id","nunique")
    ).reset_index()
    ltv_by_channel["LTV"] = ltv_by_channel["total_revenue"] / ltv_by_channel["n_users"]
    ltv_by_channel["CAC"] = channels.set_index("channel_name").loc[ltv_by_channel["channel"],"cost_per_acquisition"].values
    ltv_by_channel["ROI"] = ltv_by_channel["LTV"] / ltv_by_channel["CAC"]
    ltv_by_channel["payback_days"] = ltv_by_channel["CAC"] / (ltv_by_channel["LTV"] / ltv_window_days)
    return ltv_by_channel.sort_values("ROI", ascending=False)
```

### 渠道 LTV 分析的完整工程实现——从数据到结论到可视化

```python
def compute_channel_ltv(orders: pd.DataFrame, channels: pd.DataFrame,
                        ltv_window_days: int = 180) -> pd.DataFrame:
    """计算各渠道的 LTV（用户生命周期价值）。

    📐 口径:
        分子 = SUM(amount) WHERE order_date BETWEEN user_first_order AND user_first_order + ltv_window_days
        分母 = COUNT(DISTINCT user_id)——该渠道获取的总用户数
        时间窗口 = ltv_window_days 天——默认 180 天（6 个月）——行业标准
        为什么 180 天: 6 个月足以覆盖大部分用户的活跃周期——超过 180 天的用户已经是"稳定留存"——增量有限

    LTV = 用户在时间窗口内的总消费 / 该渠道获取的总用户数
    """
    # 每个用户的首单日期 + 渠道
    first_order = orders.groupby('user_id').agg(
        first_date=('order_date', 'min'),
        channel=('channel', 'first')
    ).reset_index()

    # 只保留首单后 ltv_window_days 内的订单
    merged = orders.merge(first_order, on='user_id')
    ltv_orders = merged[
        (merged['order_date'] >= merged['first_date']) &
        (merged['order_date'] <= merged['first_date'] + pd.Timedelta(days=ltv_window_days))
    ]

    # 按渠道汇总
    ltv_by_channel = ltv_orders.groupby('channel').agg(
        total_revenue=('amount', 'sum'),
        n_users=('user_id', 'nunique')
    ).reset_index()
    ltv_by_channel['LTV'] = ltv_by_channel['total_revenue'] / ltv_by_channel['n_users']
    ltv_by_channel['CAC'] = channels.set_index('channel_name').loc[ltv_by_channel['channel'], 'cost_per_acquisition'].values
    ltv_by_channel['ROI'] = ltv_by_channel['LTV'] / ltv_by_channel['CAC']
    ltv_by_channel['payback_days'] = ltv_by_channel['CAC'] / (ltv_by_channel['LTV'] / ltv_window_days)

    return ltv_by_channel.sort_values('ROI', ascending=False)
```

### RFM 的工程化复用——从鲜果多专用到任何订单表通用

```python
def compute_rfm(orders_df, reference_date, user_col="user_id", date_col="order_date", amount_col="amount"):
    """通用 RFM 分群——可在任何项目中复用。粒度: 一行 = 一个用户。分箱: 等频 q=4。"""
    ref_date = pd.Timestamp(reference_date)
    orders_df[date_col] = pd.to_datetime(orders_df[date_col])
    rfm = orders_df.groupby(user_col).agg(
        recency_days=(date_col, lambda x: (ref_date - x.max()).days),
        frequency=(date_col, "nunique"), monetary=(amount_col, "sum")
    ).reset_index()
    rfm["r_score"] = pd.qcut(rfm["recency_days"], q=4, labels=[4,3,2,1])
    rfm["f_score"] = pd.qcut(rfm["frequency"], q=4, labels=[1,2,3,4])
    rfm["m_score"] = pd.qcut(rfm["monetary"], q=4, labels=[1,2,3,4])
    rfm["rfm_score"] = rfm["r_score"].astype(int)*100 + rfm["f_score"].astype(int)*10 + rfm["m_score"].astype(int)
    def label_rfm(row):
        r,f,m = int(row["r_score"]), int(row["f_score"]), int(row["m_score"])
        avg = (r+f+m)/3
        if avg>=3.5: return "重要价值" if r>=3 else "重要发展" if f>=3 else "重要保持"
        elif avg>=2.5: return "重要挽留" if r<=2 else "一般价值"
        else: return "一般发展" if f>=2 else "一般保持" if r>=2 else "一般挽留"
    rfm["rfm_segment"] = rfm.apply(label_rfm, axis=1)
    return rfm
```

```python
        else: return "一般发展" if f>=2 else "一般保持" if r>=2 else "一般挽留"
    rfm["rfm_segment"] = rfm.apply(label_rfm, axis=1)
    return rfm
```

**复用验证——同一个函数不同项目只需改参数：** 鲜果多: compute_rfm(orders, "2026-09-01")。项目 1: compute_rfm(orders, "2026-12-21", user_col="customer_id", amount_col="price")。
