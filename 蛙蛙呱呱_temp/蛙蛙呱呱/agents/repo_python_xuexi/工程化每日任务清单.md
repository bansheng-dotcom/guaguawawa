```python
def check_srm(control_n: int, treatment_n: int, expected_ratio: float = 0.5) -> Dict:
    """检验分流比例是否偏离预期。

    SRM 失败 = 分流系统有问题，在分析任何指标前必须先过此检查。
    用卡方拟合优度检验，H0: 实际分流比例=预期比例。
    """
    from scipy.stats import chisquare
    total = control_n + treatment_n
    expected = [total * expected_ratio, total * (1 - expected_ratio)]
    observed = [control_n, treatment_n]
    chi2, p = chisquare(observed, f_exp=expected)
    return {"chi2": chi2, "p_value": p, "srm_detected": p < 0.01}
```

```python
def cuped_adjust(y: np.ndarray, x: np.ndarray, x_pre_exp: np.ndarray,
                 is_control: bool = False) -> tuple[np.ndarray, float]:
    """CUPED 方差缩减。θ 必须在对照组数据上计算！"""
    assert is_control, "θ 必须在对照组数据上计算，否则会引入偏差"
    theta = np.cov(y, x_pre_exp)[0, 1] / np.var(x_pre_exp)
    y_adj = y - theta * (x - x_pre_exp)
    var_reduction = np.corrcoef(y, x_pre_exp)[0, 1] ** 2
    logger.info("CUPED: 方差缩减 %.1f%%", var_reduction * 100)
    return y_adj, var_reduction
```
