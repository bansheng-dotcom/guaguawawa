### 数据契约的完整工程实现——从 YAML 到自动校验

```python
import yaml
import pandas as pd
from typing import Dict, List

def validate_against_contract(df: pd.DataFrame, contract_path: str) -> Dict:
    """对照数据契约 YAML 校验 DataFrame——自动报告所有违规。

    Args:
        df: 待校验的数据
        contract_path: data contract YAML 文件路径

    Returns:
        dict with:
            passed: bool——全部校验通过？
            total_checks: int——总共执行了多少项校验
            passed_checks: int——通过了多少项
            violations: list——每项违规的详细描述
    """
    with open(contract_path, 'r', encoding='utf-8') as f:
        contract = yaml.safe_load(f)

    violations = []
    total_checks = 0
    passed_checks = 0

    for col_def in contract['columns']:
        col_name = col_def['name']
        expected_type = col_def['type']

        # Check 1: column exists
        total_checks += 1
        if col_name not in df.columns:
            violations.append(f"[{contract['table']}] 缺失列: {col_name}")
            continue
        passed_checks += 1

        # Check 2: nullable
        if not col_def.get('nullable', True):
            total_checks += 1
            null_count = df[col_name].isna().sum()
            if null_count > 0:
                violations.append(
                    f"[{contract['table']}] {col_name}: 含 {null_count} 个 NULL"
                    f" ({null_count/len(df)*100:.1f}%)——contract 声明 nullable=false"
                )
            else:
                passed_checks += 1

        # Check 3: unique
        if col_def.get('unique', False):
            total_checks += 1
            dupes = df[col_name].duplicated().sum()
            if dupes > 0:
                violations.append(
                    f"[{contract['table']}] {col_name}: {dupes} 个重复值——"
                    f"contract 声明 unique=true"
                )
            else:
                passed_checks += 1

        # Check 4: allowed_values
        if 'allowed_values' in col_def:
            total_checks += 1
            actual_values = set(df[col_name].dropna().unique())
            invalid = actual_values - set(col_def['allowed_values'])
            if invalid:
                violations.append(
                    f"[{contract['table']}] {col_name}: 含不允许的值 {invalid}"
                )
            else:
                passed_checks += 1

        # Check 5: min/max for numeric
        if expected_type in ('int', 'float'):
            if 'min' in col_def:
                total_checks += 1
                below_min = (df[col_name] < col_def['min']).sum()
                if below_min > 0:
                    violations.append(
                        f"[{contract['table']}] {col_name}: {below_min} 行 < min({col_def['min']})"
                    )
                else:
                    passed_checks += 1
            if 'max' in col_def:
                total_checks += 1
                above_max = (df[col_name] > col_def['max']).sum()
                if above_max > 0:
                    violations.append(
                        f"[{contract['table']}] {col_name}: {above_max} 行 > max({col_def['max']})"
                    )
                else:
                    passed_checks += 1

    passed = len(violations) == 0
    return {
        'passed': passed,
        'total_checks': total_checks,
        'passed_checks': passed_checks,
        'violations': violations
    }
```

**这个函数的价值——数据到达时自动对照契约校验——任何偏离在数据进入分析管道之前就被拦下来。** 这就是 W16 的 GUIIDE 验收标准在数据层的实现——数据提供方和消费方之间的"接口契约"被代码强制执行——而不是靠"沟通"和"信任"。

import jieba
import jieba.analyse
from collections import Counter

def analyze_ugc_text(df: pd.DataFrame, text_col: str = 'text_clean') -> pd.DataFrame:
    """UGC 文本深度分析——分词+关键词+情感倾向。

    中文分词用 jieba——精确模式（默认）——适合文本分析。
    关键词提取用 TF-IDF——不需要训练——基于词频和逆文档频率。
    """
    # 分词——精确模式
    df['tokens'] = df[text_col].apply(
        lambda x: list(jieba.cut(str(x))) if x else []

    # 关键词提取——TF-IDF——Top 5
    df['keywords'] = df[text_col].apply(
        lambda x: jieba.analyse.extract_tags(str(x), topK=5) if x else []
    )

    # 词频统计——全局
    all_tokens = []
    for tokens in df['tokens']:
        all_tokens.extend([t for t in tokens if len(t) > 1])  # 过滤单字
    word_freq = Counter(all_tokens).most_common(50)

    # 文本长度分布——识别异常
    df['text_length'] = df[text_col].str.len()
    df['is_short'] = df['text_length'] < 10    # 过短——可能无意义
    df['is_long'] = df['text_length'] > 2000    # 过长——可能是爬虫抓取的全文

    short_pct = df['is_short'].sum() / len(df) * 100
    long_pct = df['is_long'].sum() / len(df) * 100

    if short_pct > 20:

    if long_pct > 10:
        logger.warning(f"长文本比例 {long_pct:.1f}%——"

### UGC 文本清洗的完整工程 pipeline

文本分析的工程化挑战——文本数据没有固定的 schema。标题可能为空、可能含 HTML 标签、可能是纯 emoji。你的文本处理 pipeline 必须处理这些边界情况。

```python
import re
import logging
from typing import List, Optional
import pandas as pd

logger = logging.getLogger(__name__)

def clean_text_pipeline(
    df: pd.DataFrame,
    text_col: str = "title",
    max_length: int = 500
) -> pd.DataFrame:
    """UGC 文本清洗 pipeline——处理 HTML/空值/emoji/长度。

    步骤:
    1. HTML 标签清洗——正则去除 <tag>
    2. 空值填充——空字符串而不是 NaN（NaN 会导致后续字符串操作报错）
    3. emoji 保留或移除（默认保留——emoji 可能蕴含情感信息）
    4. 长度计算——新增 text_length 列
    5. 异常检测——空文本比例过高时 logger.warning

    Returns:
        DataFrame with added columns: text_clean, text_length, is_empty
    """
    total = len(df)

    # Step 1: HTML 标签清洗
    df['text_clean'] = df[text_col].fillna('').str.replace(
        r'<[^>]*>', '', regex=True
    )

    # Step 2: 空值填充 + strip
    df['text_clean'] = df['text_clean'].fillna('').str.strip()

    # Step 3: emoji 保留（默认——可通过参数控制移除）
    # emoji 模式: [\U0001F600-\U0001F64F\U0001F300-\U0001F5FF...]
    # 保留不做处理——只记录含 emoji 的比例
    emoji_pattern = re.compile(r'[\U0001F300-\U0001F9FF]')
    has_emoji = df['text_clean'].apply(lambda x: bool(emoji_pattern.search(str(x))))
    emoji_pct = has_emoji.sum() / total * 100

    # Step 4: 长度计算
    df['text_length'] = df['text_clean'].str.len()
    df['is_empty'] = df['text_length'] == 0

    # Step 5: 异常检测
    empty_pct = df['is_empty'].sum() / total * 100
    if empty_pct > 30:
        logger.warning(
            f"空文本比例 {empty_pct:.1f}% ({df['is_empty'].sum()}/{total})——"
            f"数据源可能有问题——建议检查上游数据管道"
        )
    if emoji_pct > 50:
        logger.info(f"含 emoji 文本比例 {emoji_pct:.1f}%——内容可能偏向社交/娱乐")

    logger.info(
        f"文本清洗完成: {total} 行——空文本 {empty_pct:.1f}%——"
        f"平均长度 {df['text_length'].mean():.0f} 字符——含 emoji {emoji_pct:.1f}%"
    )

    return df
```

**这个 pipeline 的工程化价值——不是"能跑"——是"边跑边报告数据质量"。** 如果某天 50% 的标题突然为空——logger.warning 会在日志中留下记录——你不会在分析报告已经发出后才发现数据有问题。

**面试官一定会追问的 3 个问题——提前准备标准回答:**

**追问 1:** "你的内容质量评分——权重怎么定的？主观吗？"

> **A:** "不是主观的。我的初始权重基于历史数据——用线性回归以'用户 7 天留存'为目标变量——各维度的标准化回归系数作为初始权重。完读率的系数最高（0.35）——这和业务直觉一致——用户完整读完的内容——质量通常更高。留存贡献的系数次高（0.30）——这反映了内容的长期价值——用户读完这篇内容后愿意回访——说明内容对平台有持续的吸引力。后续经业务团队讨论——微调了 engagement_rate 从 0.12 到 0.15——因为业务认为'互动'在冷启动阶段比'完读'更重要。目前的权重是'数据驱动+业务共识'的产物——不是拍脑袋——也不是纯数据（因为数据有噪声和局限）。"

**追问 2:** "你的评分体系有什么局限？"

> **A:** "三个局限。(1) 时效性偏差——新发布的内容因为曝光量少——互动率/完读率/分享率都不稳定——评分波动大。解决方案——对新内容引入'贝叶斯平滑'——用先验均值拉低极端值。(2) 内容类型偏差——短视频和长文章的'好'的标准不同——用同一套权重会不公平。解决方案——按内容类型分组——每个组独立拟合权重。(3) 马太效应——已经获得高曝光的内容因为曝光量大——互动/完读/分享的基数大——评分会越来越高——形成'富者越富'的循环。解决方案——在推荐权重中引入'探索因子'——给低曝光高质量候选内容额外的曝光机会。"

**追问 3:** "如果让你重新设计——你会怎么改进？"

> **A:** "两点。(1) 引入时间衰减——最近 30 天的互动比 90 天前的互动更重要——用指数加权移动平均替代简单平均。(2) 引入用户分层——不同用户群对'好内容'的定义不同——新用户可能偏好短内容——老用户可能偏好深度长文——按用户群分别训练评分模型——而不是一刀切。"

```python
def compute_content_quality_score(
    df: pd.DataFrame,
    weights: dict = None,
    normalize: bool = True
) -> pd.DataFrame:
    """计算内容质量指数（0-100 分）——综合多维度加权归一化。

    维度:
    - engagement_rate: 互动率 = (点赞+评论+分享+收藏) / 曝光量
    - completion_rate: 完读率 = 完整阅读用户数 / 点击用户数
    - share_rate: 分享率 = 分享用户数 / 点击用户数
    - retention_contribution: 留存贡献——读完该内容的用户中——7 天后回访的比例
    - freshness: 时效性——发布时间距今的天数（越新越高）

    权重来源（在 docstring 中声明——不是"我拍的"）:
    当前权重基于历史数据用线性回归拟合——以"用户 7 天留存"为目标变量——
    各维度的标准化系数作为初始权重。后续经业务团队讨论微调。
    如需修改权重——在 PR 中附上数据依据——不可随意更改。

    Args:
        df: 含各维度原始值的内容 DataFrame
        weights: 各维度的权重字典——默认等权
        normalize: 是否将各维度归一化到 [0,1] 后再加权

    Returns:
        DataFrame with added columns: quality_score, score_rank, score_tier(S/A/B/C/D)
    """
    if weights is None:
        weights = {
            'engagement_rate': 0.15,
            'completion_rate': 0.35,  # 完读率权重最高——最能反映内容质量
            'share_rate': 0.15,
            'retention_contribution': 0.30,  # 留存贡献次高——反映长期价值
            'freshness': 0.05
        }

    # 归一化——Min-Max 到 [0,1]
    scores = pd.DataFrame(index=df.index)
    for col in weights:
        if col in df.columns:
            if normalize:
                col_min, col_max = df[col].min(), df[col].max()
                if col_max > col_min:
                    scores[col] = (df[col] - col_min) / (col_max - col_min)
                else:
                    scores[col] = 0.5
            else:
                scores[col] = df[col]

    # 加权求和——缩放到 0-100
    df['quality_score'] = sum(
        scores[col] * weight for col, weight in weights.items() if col in scores.columns
    ) * 100

    # 排名 + 分层
    df['score_rank'] = df['quality_score'].rank(ascending=False).astype(int)
    df['score_tier'] = pd.cut(
        df['quality_score'],
        bins=[0, 30, 50, 70, 85, 101],
        labels=['D', 'C', 'B', 'A', 'S']
    )

    return df.sort_values('quality_score', ascending=False)
```
